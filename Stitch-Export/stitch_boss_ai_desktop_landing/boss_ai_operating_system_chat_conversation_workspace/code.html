I've identified the clock-skew vulnerability in verifyJwtToken(). When tokens are validated near expiration boundaries, server drift can cause premature invalidation. Here is the patched middleware with a configurable skew window (defaulting to 60 seconds) and safe timestamp arithmetic:

src/auth/jwt-middleware.ts

export interface JwtVerifyOptions {
  clockToleranceSeconds?: number; // Default 60s skew allowance
  issuer?: string;
}

export function isTokenExpired(
  expSeconds: number,
  tolerance: number = 60
): boolean {
  const currentTimestamp = Math.floor(Date.now() / 1000);
  // Allow token validity if within current time minus tolerance window
  return currentTimestamp >= (expSeconds + tolerance);
}

export async function authMiddleware(req: Request, opts: JwtVerifyOptions = {}) {
  const token = req.headers.get('authorization')?.replace(/^Bearer\s+/, '');
  if (!token) throw new AuthError('MISSING_TOKEN');
  
  const payload = await decodeAndVerify(token, opts);
  if (isTokenExpired(payload.exp, opts.clockToleranceSeconds)) {
    throw new AuthError('TOKEN_EXPIRED_CLOCK_SKEW_EXCEEDED');
  }
  return payload;
}
